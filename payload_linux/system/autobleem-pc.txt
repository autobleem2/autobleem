# AutoBleem first-boot options - the PC USB stick.
#
# This file sits on the stick's small first partition (FAT), so it can be edited from any Windows/macOS/
# Linux machine before the first boot. AutoBleem's setup reads it once, on the boot that installs
# AutoBleem, and turns each key into an installer option. A key that is missing or commented out keeps its
# default. Lines starting with # are comments.
#
# WiFi is not set here: with no network cable plugged in, the first boot asks for a WiFi network on the screen.

# How the setup is shown. gfx draws its screens graphically (the AutoBleem logo with the questions and the
# progress bars under it); text draws the same screens as text on the console, raspi-config's look - for a
# PC where the graphical screen stays black or garbled. Left unset, the first boot asks (graphical after
# 30 seconds with no answer). The choice is kept in /etc/autobleem/installer-ui on the Linux partition, and
# the launcher's online updates draw with the same screen.
#installer=gfx

# Size of the Linux system partition, in GiB. Everything else on the stick becomes the AUTOBLEEM games
# partition (exFAT - plug the stick into any computer and drop games on it). 8 is plenty. 0 leaves the
# partition as it is.
root_gib=8

# RetroArch - optional. AutoBleem plays PlayStation games on its own; RetroArch adds the other systems (NES,
# SNES, Mega Drive, arcade, ...) at the cost of close to a GB of downloads. Left unset, the first boot asks
# on the screen (no answer within a minute = yes). Set it to skip the question:
#   prebuilt - the latest release from AutoBleem's download site (seconds; built here if the site is down)
#   source - latest release, built on this PC (10-40 minutes) | apt - the distribution's package
#   none - PS1-only AutoBleem
#retroarch=prebuilt

# AutoBleem's download site (the prebuilt RetroArch). Only for a mirror or a copy on your own network.
#repo=https://autobleem.retromenele.pl

# PS1 box art. The launcher fetches each game's cover from thumbnails.libretro.com when it scans the
# game, so nothing needs mirroring: none (default) | boxarts (the whole set, ~350 MB, for covers offline) |
# all (title screens and snaps too)
#thumbnails=none

# The BIOS pack (RetroArch/system, and the PS1 BIOS for pcsx-ab): yes | no. Without RetroArch only the two
# PS1 files are fetched.
#bios=yes

# RetroArch cores, core info, menu assets, databases from buildbot.libretro.com (a few hundred MB): yes | no.
# Ignored (no) without RetroArch.
#downloads=yes

# Sample games (1 MB from AutoBleem's download site): with RetroArch a NES, two Super NES and a Mega Drive
# homebrew game, each with its box art - so there is something on the shelf before you copy your own games
# over. SAMPLES.md on the games partition names them and their licences; delete any you do not want. yes | no
#samples=yes
